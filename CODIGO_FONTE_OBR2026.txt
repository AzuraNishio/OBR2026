class line_follower_constants:

    # Maximum linear speed for the sensor in mm/s
    max_sensor_linear_speed = 160

    # Reduced linear speed for the sensor for more precise control in mm/s
    sensor_reduced_linear_speed = 80

    # Duration for the sensor linear speed ramp-up and ramp-down
    sensor_linear_speed_ramp_duration = 0.5

    # Constants for the gap check maneuver in mm
    gap_check_distance = 250
    do_gap_save = False

    do_knock_down_save = False

    # Minimum value for the silver detection in %
    silver_min_threshold = 95

    # Side to go around the obstacle, -1 for left, 1 for right
    obstacle_side = -1
    do_obstacle = True
    obstacle_parallel_dist = 450
    obstacle_trigger_dist = 60

    # PD controller coefficients
    KP = 4.0
    KD = 1.0
    KD2 = 0.1

    # Line follower bias to compensate for sensor misalignment or other factors
    bias = -4

ALT_FLAG      = 0xA0;
STOP_FLAG     = 0xA5;
ALT_ALT_FLAG  = 0xA1;
ALT_STOP_FLAG = 0xA6;


class ESP32_bridge:
    def __init__(self, port):
        self.port = port
        self.device = UARTDevice(port, baudrate=115200, power_pin=1)
        self.packet_buffer = [[] for _ in range(8)]
        self.active_packet = 0
        self.is_alted = False
        self.unalt_map = {ALT_ALT_FLAG: ALT_FLAG, ALT_STOP_FLAG: STOP_FLAG}
        self.alt_map = {ALT_FLAG: [ALT_FLAG, ALT_ALT_FLAG], STOP_FLAG: [ALT_FLAG, ALT_STOP_FLAG]}
        self._avaiable = False

    def avaiable(self): return self._avaiable

    def get_packet(self):
        self._avaiable = False
        return self.packet_buffer[self.active_packet]

    def _write_to_packet_buffer(self, byte):
        self.packet_buffer[(self.active_packet + 1) % 8].append(byte)

    def _end_current_packet(self):
        self.is_alted = False
        self.active_packet = (self.active_packet + 1) % 8
        self.packet_buffer[(self.active_packet + 1) % 8] = []
        self._avaiable = True

    async def tick_packet_receiver(self):
        while self.device.waiting():
            byte = (await self.device.read(1))[0]
            if self.is_alted:
                self._write_to_packet_buffer(self.unalt_map.get(byte, 0x00))
                self.is_alted = False
            else:
                if byte == ALT_FLAG:
                    self.is_alted = True
                elif byte == STOP_FLAG:
                    self._end_current_packet()
                else:
                    self._write_to_packet_buffer(byte)

class ESP32_distance_sensor(ESP32_bridge):
    def __init__(self, port):
        super().__init__(port)

    async def distance(self):
        await self.tick_packet_receiver()
        if len(self.get_packet()) < 1: return 9999
        if len(self.get_packet()) > 0 and self.get_packet()[0] == 0: return 9999
        return (self.get_packet()[0] | (self.get_packet()[1] << 8))

    
    async def L_button_pressed(self):
        await self.tick_packet_receiver()
        if len(self.get_packet()) < 1: return 9999
        if len(self.get_packet()) > 0 and self.get_packet()[0] == 0: return 9999
        return False # self.get_packet()[3] == 1

    async def R_button_pressed(self):
        await self.tick_packet_receiver()
        if len(self.get_packet()) < 1: return 9999
        if len(self.get_packet()) > 0 and self.get_packet()[0] == 0: return 9999
        return self.get_packet()[4] == 1

class FollowerAlgorithm:
    def __init__(self, ctx: Context, model: TreadmilDifferentialDriveRobotModel, robot: LineFollowerRobot):
        self.ctx = ctx
        self.robot = robot
        self.model = model

    # region Setup

    def setup(self):
        ctx = self.ctx
        robot = self.robot
        model = self.model

        ctx.imu.reset_heading(0)

        self.R_green_counter = 0
        self.L_green_counter = 0
        self.tps = 500
        self.tps_counter = 0
        self.last_tps_check = ctx.time()
        # initialize sensor linear speed to configured max
        self.sensor_linear_speed = line_follower_constants.max_sensor_linear_speed

        # runtime state
        # `sensor_linear_speed` initialized to configured max

        self.last_time = self.ctx.time()
        self.lastCheck = 0
        self.last_error = 0
        self.last_derivative = 0
        self.garra_was_down = False

        #robot.drive_base.use_gyro(True)

    # endregion

    # region Main Loop

    async def tick(self):
        self.tps_counter += 1
        dT = self.ctx.time() - self.last_time
        self.last_time = self.ctx.time()
        if (0.5 < self.ctx.time() - self.last_tps_check):
            self.tps = ((self.tps_counter * 2) + (self.tps * 4)) * 0.2
            self.tps_counter = 0
            self.last_tps_check = self.ctx.time()

        await self.fuzzy_loop(max(0.1, dT))

    def freeze_tps(self):
        self.interval = self.ctx.time() - self.last_tps_check

    def unfreeze_tps(self):
        self.last_tps_check = self.ctx.time() - self.interval

    async def fuzzy_loop(self, dT):
        robot = self.robot

        await wait(1)

        R_reflection, L_reflection, darkest, R_green, L_green = await self.read_sensors()

        self.update_green_counters(dT, R_green, L_green)

        if line_follower_constants.do_knock_down_save:await self.handle_claw_tilt()

        # Distance-based gap detection (replaces the old cruise-mode gap detection)
        if line_follower_constants.do_gap_save: await self.handle_gap_detection()

        if R_reflection < 50 or L_reflection < 50:
            self.lastCheck = robot.drive_base.distance()

        if line_follower_constants.do_obstacle: await self.handle_obstacle_avoidance()

        if darkest > line_follower_constants.silver_min_threshold:
            await room3check(robot, self.ctx)

        if darkest < 11: # Check wether a sensor detected black. If so run green check and crossing check
            self.freeze_tps()  # Freeze tick per second counter

            # If either sensor has seen green less than 0.1 seconds ago, do green
            if self.L_green_counter < 0.1 or self.R_green_counter < 0.1: 
                await self.green()
            else: # Else do center aligned manuever
                await self.center_aligned_manuever(sign(L_reflection - R_reflection))

            self.unfreeze_tps()

        await self.line_follow_control(dT, R_reflection, L_reflection)

    # endregion

    # region Sensor Reading

    async def read_sensors(self):
        robot = self.robot

        R_reflection = await robot.R_sensor.reflection()
        L_reflection = await robot.L_sensor.reflection()

        darkest = min(R_reflection, L_reflection)

        R_green = await robot.R_sensor.color() == Color.GREEN
        L_green = await robot.L_sensor.color() == Color.GREEN

        return R_reflection, L_reflection, darkest, R_green, L_green

    def update_green_counters(self, dT, R_green, L_green):
        self.R_green_counter = 0 if R_green else min(20, self.R_green_counter + dT)
        self.L_green_counter = 0 if L_green else min(20, self.L_green_counter + dT)

    # endregion

    # region Claw / Tilt Handling

    async def handle_claw_tilt(self):
        ctx = self.ctx
        robot = self.robot

        if abs(ctx.imu.tilt()[0]) > 27:
            if not self.garra_was_down:
                self.garra_down_time = ctx.time()
            self.garra_was_down = True

            if not abs(ctx.time() - self.garra_down_time) > 2:
                robot.garra_motor.run_time(5000, 1000)
                await robot.drive_base.straight(70)
            else:
                await robot.garra_motor.run_time(-200, 1000)
                await robot.drive_base.straight(70)
        else:
            self.garra_was_down = False
            robot.garra_motor.run_time(-200, 1000)

    # endregion

    # region Gap Detection (distance based)

    async def handle_gap_detection(self):
        robot = self.robot

        if robot.drive_base.distance() - self.lastCheck <= line_follower_constants.gap_check_distance:
            return

        robot.drive_base.use_gyro(True)
        robot.drive_base.curve(0, 20)
        isLine = False
        while not robot.drive_base.done():
            isLine = isLine or await robot.R_sensor.reflection() < 30 or await robot.L_sensor.reflection() < 30
        robot.drive_base.brake()

        if not isLine:
            await robot.drive_base.curve(40, -10)
            for i in range(4):
                start_dist = robot.drive_base.distance()
                while True:
                    robot.drive(-500, 0)
                    if abs(robot.drive_base.distance() - start_dist) > (10 + line_follower_constants.gap_check_distance) / 4:
                        break
                    if await robot.is_seeing_any_black():
                        break

                await robot.turn(-20)
                bb = await robot.is_seeing_any_black()
                await robot.turn(20)
                if bb: break
            await robot.turn(await robot.black_dir() * 6)
            await robot.straight(-60)

            
        else:
            await robot.drive_base.curve(0, -20)

        robot.drive_base.use_gyro(False)

        self.lastCheck = robot.drive_base.distance()

    # endregion

    # region Obstacle Avoidance

    async def handle_obstacle_avoidance(self):
        robot = self.robot
        l = line_follower_constants.obstacle_side

        distance = await robot.esp32_sensor_pack.laser_distance()

        if distance >= line_follower_constants.obstacle_trigger_dist:
            return
        
        self.robot.brake()

        # Surprise challenge (ignore)
        if surprises.signal_on_obstacle:
            await self.ctx.wait(6)
            await robot.signal_blink()

        await robot.drive_base.straight(distance - 20)
        await robot.drive_base.straight(-40)

        robot.drive_base.use_gyro(True)
        await robot.drive_base.straight(-10)

        await robot.drive_base.turn(l * 90)

        await robot.drive_base.straight(-100)
        await robot.alignOnLinePerTime(3)
        await robot.drive_base.straight(150)

        while True:
            await robot.drive_base.turn(l * -90)
            if await robot.esp32_sensor_pack.laser_distance() > 80:
                await robot.drive_base.turn(l * 90)
                await robot.drive_base.straight(110)
                await robot.drive_base.turn(l * -90)
                break
            else:
                await robot.drive_base.turn(l * 90)
                await robot.drive_base.straight(60)

        await robot.drive_base.straight(line_follower_constants.obstacle_parallel_dist)
        await robot.drive_base.turn(l * -90)
        await robot.drive_base.straight(40)


        while await robot.R_sensor.reflection() > 30:
            robot.drive_base.drive(100, 0)

        await robot.drive_base.straight(70)
        await robot.drive_base.turn(l * 90)

        robot.drive_base.use_gyro(False)
        self.lastCheck = robot.drive_base.distance()

    # endregion

    # region Green Crossing Handling

    async def green(self):
        robot = self.robot

        # Check side; 1 if right, 0 if 180, -1 if left
        side = (1 if self.R_green_counter < 0.1 else 0) - (1 if self.L_green_counter < 0.1 else 0)
        if side > 0:
            pro_line_sensor = robot.R_sensor
            retro_line_sensor = robot.L_sensor
        else:
            pro_line_sensor = robot.L_sensor
            retro_line_sensor = robot.R_sensor

        print(f"green! {side}")

        # First of all brake
        robot.drive_base.brake()

        robot.drive_base.use_gyro(True)
        # After braking, move forward 70 mm
        await robot.drive_base.straight(70)

        if side == 0:  # If 180 them turn 170
            await robot.drive_base.curve(0, -90)
            await robot.straight(15)
            await robot.drive_base.curve(0, -80)
            side = -1  # Sets side to -1 so turn till line works
        else:  # Else turn 80 in the direction
            await robot.drive_base.curve(0, 60 * side)

        # Afterwards keep turning untill it finds the line
        await self._green_turn_until_line(pro_line_sensor, retro_line_sensor, side)

        robot.drive_base.brake()

        robot.drive_base.use_gyro(False)

    async def _green_turn_until_line(self, pro_line_sensor, retro_line_sensor, side):
        robot = self.robot

        skip = False
        flip = False
        start_angle = self.ctx.imu.heading()
        while True and not skip:  # Loop until skip
            robot.drive_base.drive(0, side * 140)  # Just turn in the direction of the line
            if await pro_line_sensor.reflection() < 21:
                while True:
                    robot.drive_base.drive(0, side * 40)

                    if await pro_line_sensor.reflection() > 21:
                        if await retro_line_sensor.reflection() > 21:
                            skip = True
                            break
                        else:
                            robot.drive_base.curve(0, side * -4)
            elif await retro_line_sensor.reflection() < 21:
                while True:
                    robot.drive_base.drive(0, side * -40)

                    if await retro_line_sensor.reflection() > 21:
                        if await pro_line_sensor.reflection() > 21:
                            skip = True
                            break
                        else:
                            robot.drive_base.curve(0, side * 4)

            if abs(self.ctx.imu.heading() - start_angle) > 50:
                flip = True
                await robot.drive_base.turn(side * -40)
                break

        if flip:
            await self._green_turn_until_line(retro_line_sensor, pro_line_sensor, -side)

    # endregion

    # region Line Follow Control

    async def line_follow_control(self, dT, R_reflection, L_reflection):
        ctx = self.ctx
        robot = self.robot
        model = self.model

        # Robot turning speed
        turning_speed = ctx.imu.angular_velocity(vector(0, 0, 1))

        # Calculate turning rate from error
        error = L_reflection - R_reflection + line_follower_constants.bias
        derivative = (error - self.last_error) / dT
        second_derivative = (derivative - self.last_derivative) / dT
        self.last_error = error
        self.last_derivative = derivative

        # PD controller using gains from constants
        turn_rate = (error * line_follower_constants.KP) + (derivative * line_follower_constants.KD) + (second_derivative * line_follower_constants.KD2)

        if abs(turn_rate) < 10:
            turn_rate = 0

        sensor_linear_speed = self._ramp_linear_speed(dT)

        # Calculate maximum turn rate
        max_turn_rate = sensor_linear_speed / model.sensor_turn_rate_to_linear

        # If too fast, clamp turn rate
        turn_rate = clamp(max_turn_rate, -max_turn_rate, turn_rate)

        # Calculate sideways sensor speed and planned sideways sensor speed
        sideways_sensor_speed = model.sensor_turn_rate_to_linear * turning_speed
        planed_sideways_sensor_speed = model.sensor_turn_rate_to_linear * turn_rate

        # Afterwards calculate linear speed accordingly to sideways planned and real sensor speed to keep sensor under speed limit
        linear = sensor_linear_speed - max(abs(sideways_sensor_speed), abs(planed_sideways_sensor_speed))

        robot.drive_base.drive(linear, turn_rate)

    def _ramp_linear_speed(self, dT):
        sensor_linear_speed = self.sensor_linear_speed
        if self.sensor_linear_speed < line_follower_constants.max_sensor_linear_speed:
            self.sensor_linear_speed += dT * line_follower_constants.max_sensor_linear_speed / line_follower_constants.sensor_linear_speed_ramp_duration
        else:
            self.sensor_linear_speed = line_follower_constants.max_sensor_linear_speed
        return sensor_linear_speed

    # endregion

    # region Center Aligned Manuever

    async def center_aligned_manuever(self, dir):
        self.sensor_linear_speed = line_follower_constants.sensor_reduced_linear_speed
        print("Center aligned manuever")
        robot = self.robot

        await robot.drive_base.straight(5)
        if self.robot.R_sensor.color() == Color.BLACK and self.robot.L_sensor.color() == Color.BLACK:
            robot.drive_base.straight(20)
            return

        if dir < 0:
            pro_line_sensor = robot.L_sensor
            ret_line_sensor = robot.R_sensor
        else:
            pro_line_sensor = robot.R_sensor
            ret_line_sensor = robot.L_sensor

        skip, pro_line_width, ret_line_width = await self._center_align_rotation(dir, pro_line_sensor, ret_line_sensor)

        print(f"Pro line: {pro_line_width}mm, ret line: {ret_line_width}mm")

        if not skip:
            await self._center_turn_until_line(dir, pro_line_sensor, ret_line_sensor)

    async def _center_align_rotation(self, dir, pro_line_sensor, ret_line_sensor):
        robot = self.robot

        # Step to align CoR
        robot.drive_base.curve(110, -25 * dir, wait=False)

        last_dist = robot.drive_base.distance()

        pro_line_width = 0
        ret_line_width = 0

        post_line = 0

        skip = False

        while not robot.drive_base.done():
            dP = robot.drive_base.distance() - last_dist
            last_dist = robot.drive_base.distance()
            if await pro_line_sensor.reflection() < 8:
                pro_line_width += dP
            if await ret_line_sensor.reflection() < 8:
                ret_line_width += dP

            if pro_line_width > 7 and ret_line_width > 7:
                robot.drive_base.stop()
                skip = True
                break

            if pro_line_width > 7:
                post_line += dP

            if post_line > 20:
                break

        return skip, pro_line_width, ret_line_width

    async def _center_turn_until_line(self, dir, pro_line_sensor, ret_line_sensor):
        robot = self.robot

        init_angle = robot.drive_base.angle()
        skip = False

        while True and not skip:  # Loop until skip
            robot.drive_base.drive(0, dir * 140)  # Just turn in the direction of the line

            if abs(robot.drive_base.angle() - init_angle) > 160:  # If has turned more than 160 degrees
                dir *= -1  # Them flip direction
                await robot.drive_base.turn(dir * 70)  # Turn untill its kinda of aligned to the line
                await robot.drive_base.straight(-9)  # Move back 9 mm
                while abs(robot.drive_base.angle() - init_angle) > 160:  # And rotate untill it has undone 20 degrees of its rotation
                    robot.drive_base.drive(0, dir * 140)

            if await pro_line_sensor.reflection() < 21:  # Found line with pro line sensor: spin untill pro line has passed, then set skip
                while True:
                    robot.drive_base.drive(0, dir * 70)

                    if await pro_line_sensor.reflection() > 21:
                        skip = True
                        break

            if await ret_line_sensor.reflection() < 21:  # Found line with retro line: spin opposite untill exited line, then set skip
                while True:
                    robot.drive_base.drive(0, dir * -70)

                    if await ret_line_sensor.reflection() > 21:
                        skip = True
                        break

class LineFollowerRobot:
    def __getattr__(self, name):
        """Forward attributes not defined here to the drive base."""
        return getattr(self.drive_base, name)

    async def straight(self, distance, speed = None):
        if not speed is None:
            self.drive_base.settings(straight_speed = speed)
        else:
            self.drive_base.settings(straight_speed = self.modified_straight_speed or self.default_straight_speed, straight_acceleration= 700)

        return await self.drive_base.straight(distance)

    async def turn(self, angle):
        return await self.drive_base.turn(angle + copysign(1, angle))

    def __init__(self, ctx: Context, L_sensor_port, R_sensor_port, L_motor_port, R_motor_port, wheel_diameter, wheel_distance, X_distance_CoR_sensor, Y_distance_CoR_sensor, R_motor_forward_direction = Direction.CLOCKWISE, L_motor_forward_direction = Direction.CLOCKWISE, esp32_sensor_pack = "E", garra_motor = "F"):
        self.garra_up = False
        # Keep trying to connect
        self.animTime = ctx.time()

        self.R_motor_port = R_motor_port
        self.L_motor_port = L_motor_port

        self.ctx = ctx

        while True:
            try:
                self.R_motor:Motor = ctx.get_port_of_type(R_motor_port, Motor)

class TreadmilDifferentialDriveRobotModel:
    def __init__(self, ctx: Context, robot: LineFollowerRobot):
        self.ctx = ctx
        self.robot = robot
        self.position = vector(0, 0, 0)
        self.last_time = ctx.time()
        self.last_R_wheel_theta = robot.R_motor.angle()
        self.last_L_wheel_theta = robot.L_motor.angle()

        # Constants
        self.gravity_vector_world_space = vector(0, 0, 9810) #gravity in mm/sÂ²
        self.degree_to_radian = pi / 180
        self.degree_to_wheel_linear_constant = robot.wheel_diameter * self.degree_to_radian * 0.5
        self.sensor_CoR_distance = length(vector(robot.X_distance_CoR_sensor, robot.Y_distance_CoR_sensor))
        self.sensor_turn_rate_to_linear = self.sensor_CoR_distance * self.degree_to_radian


    def acceleration_gravity_corrected_world_frame(self):
        acceleration = self.ctx.imu.orientation() * self.ctx.imu.acceleration();
        acceleration -=  self.gravity_vector_world_space
        return acceleration

    def wheel_theta_to_linear(self, theta):
        return self.degree_to_wheel_linear_constant * theta

    async def routine(self):
        while True:
            try:
                self.tick()
                await wait(10)
            except Exception as e:
                print(e)
    
    def pos_2d(self):
        return vector(self.position[0], self.position[1])
    
    def orientation_2d(self):
        orientation: Matrix = self.ctx.imu.orientation()
        v = orientation * vector(1, 0, 0)
        v = vector(v[0], v[1])
        v /= length(v)
        return v
    
    def sensor_linear_speed(self, linear_rate, turn_rate):
        rotation_wise_linear = self.sensor_turn_rate_to_linear * (turn_rate)
        return abs(rotation_wise_linear) + abs(linear_rate)


    def tick(self):
        ctx:Context = self.ctx
        dt = max(ctx.time() - self.last_time, 0.001)
        self.last_time = ctx.time()

        self.accel:Matrix = self.acceleration_gravity_corrected_world_frame()

        R_wheel_theta = self.robot.R_motor.angle()
        L_wheel_theta = self.robot.L_motor.angle()

        R_wheel_delta_theta = R_wheel_theta - self.last_R_wheel_theta
        L_wheel_delta_theta = L_wheel_theta - self.last_L_wheel_theta
        
        self.last_R_wheel_theta = R_wheel_theta
        self.last_L_wheel_theta = L_wheel_theta

        self.L_wheel_linear_speed = self.wheel_theta_to_linear(L_wheel_delta_theta) / dt
        self.R_wheel_linear_speed = self.wheel_theta_to_linear(R_wheel_delta_theta) / dt

        self.position += self.ctx.imu.orientation() * vector((self.L_wheel_linear_speed + self.R_wheel_linear_speed ) * 0.5 * dt, 0, 0)

                await robot.ctx.speaker.beep()

            break
        if await robot.is_seeing_any_black():
            data.markOutput(n)
            base.brake()

            await robot.ctx.wait(0.2)

            await robot.garraUp()
            await base.straight(data.U * 0.74)

    base.brake()
    base.use_gyro(True)
    return abs(base.distance() - start_dist) + 20 , triangle

async def align_at_wall_if_close(robot: LineFollowerRobot, wall = None, reposition = True, max_dist = 150):
    # Generic "nudge to wall if within range" helper - position depends on caller,
    # see the diagram at each call site.
    if wall:
        print(f"Aligning at wall {wall}")
    start_dist = await robot.esp32_sensor_pack.laser_distance()
    if start_dist < max_dist and not robot.data.output == wall:
        robot.use_gyro(False)
        await robot.straight(max_dist + 30)
        robot.use_gyro(True)
        if reposition:
            await robot.straight(-start_dist * 0.9)

async def align_at_wall_or_black_if_close(robot: LineFollowerRobot, wall = None, reposition = True):

    if wall:
        print(f"Aligning at wall {wall}")
    start_dist = await robot.esp32_sensor_pack.laser_distance()
    if start_dist < 150:
        robot.use_gyro(False)
        await robot.straight(170)
        robot.use_gyro(True)
        if reposition:
            await robot.straight(-start_dist * 0.9)
    elif wall == robot.data.output:
        robot.straight(50)
        while not robot.drive_base.done():
            if await robot.is_seeing_any_black():
                break
        robot.brake()
        await robot.alignOnLinePerTime(3)

async def aligned_on_triangle_deliver(robot: LineFollowerRobot, data: Room3info):

    triangles_aligned = (
        (data.triangle1 in ["0-1", "6-7"] and data.triangle2 in ["0-1", "6-7"])
        or (data.triangle1 in ["3-4", "9-10"] and data.triangle2 in ["3-4", "9-10"])
    )

    await robot.garraDown()
    await robot.move_time(2, -500)
    await robot.garraUp()
    await robot.turn(180)

    await robot.garraDown()

    if triangles_aligned:
        await robot.move_time(3, -500)
        await robot.garraUp()
        await robot.garraDown()
        await robot.straight(2 * data.U)

    await robot.move_time(2)
    await robot.straight(-data.U * 0.3, speed = 600)
    await robot.garraUp()
    await robot.garraDown()
    robot.drive_base.use_gyro(False)
    await robot.move_time(0.6)

    robot.drive_base.use_gyro(True)
    await robot.straight(-data.U * 0.5)

    await robot.straight(-data.U * 0.7)
    await robot.garraUp()

        await robot.turn(-45)
    else:  # If none do 9-10

    if data.output in [12, 4]:
        await robot.straight(-data.U * 1.3)
    if data.output in [10, 6]:
        await robot.straight(data.U * 1.3)

    if data.output in [4, 5, 6, 3, 7]:
        await robot.turn(90)                 # Facing ---->

        await robot.turn(-90)

        await robot.turn(180)

           11X|   -   |X5

           11X|       |X5  at B if out 7

class waiter:
    def __init__(self, ctx, duration, inverted = False):
        self.ctx = ctx
        self.inverted = inverted
        self.end_time = ctx.time() + duration

    def __bool__(self):
        return not self.inverted if self.ctx.time() > self.end_time else self.inverted


class Context:
    def __init__(self, errorHandler: FailHandler):
        self.state = "initializing context"

        self.errorHandler = errorHandler
        self.hub = PrimeHub()
        self.speaker = self.hub.speaker
        self.imu = self.hub.imu
        self.display = self.hub.display
        self.buttons = self.hub.buttons
        self.battery = self.hub.battery
        self.cron = StopWatch()
        self.display.off()
        self.robot = None
        errorHandler.bindContext(self)
        print("\033[2J\033[H", end="")
        print("\n")
        self.setup_ports()
        self.hub_info_dump()

    async def wait(self, time):
        start_time = self.time()
        while True:
            if self.time() - start_time > time:
                break
    
    def time(self):
        return self.cron.time() * 0.001
    
    def throw(self, e: Exception):
        self.errorHandler.exception(e)
    
    def time_ms(self):
        return self.cron.time()

    def get_waiter(self, duration):
        return waiter(self, duration)

    def get_invereted_waiter(self, duration):
        return waiter(self, duration, inverted = True)

    def setup_one_port(self, port: Port):
        pp = connect_port_device(port)
        self.ports[str(port)] = pp
        return pp
    
    def reload_one_port(self, port: Port):
        pp = connect_port_device(port)
        if pp[1] != None:
            self.ports[str(port)] = pp
            return pp
        else:
            return self.ports[str(port)]
        
    def setup_ports(self):
        self.ports = {}

        self.pA = self.setup_one_port(Port.A)
        self.pB = self.setup_one_port(Port.B)
        self.pC = self.setup_one_port(Port.C)
        self.pD = self.setup_one_port(Port.D)
        self.pE = self.setup_one_port(Port.E)
        self.pF = self.setup_one_port(Port.F)
    
    def reload_empty_ports(self):
        self.pA = self.reload_one_port(Port.A)
        self.pB = self.reload_one_port(Port.B)
        self.pC = self.reload_one_port(Port.C)
        self.pD = self.reload_one_port(Port.D)

async def room3_algorithm_speed_test(iterations=1, max_steps=16):

    from V3.r3_algorithm import algorithm, destroy_algorithm
    from V3.r3_model import all_room_descriptors, room_supplier
    from V3.r3_model_robot import r3_model_robot
    from V3.NekoMath import PrettyPrint

    if iterations < 1 or max_steps < 1:
        raise ValueError("iterations and max_steps must be positive")

    descriptors = all_room_descriptors()
    timer = StopWatch()
    total_ms = 0
    total_steps = 0
    finished_runs = 0

    pp = PrettyPrint()

    pp + "Room 3 speed test: simulated robot only; hardware will not move"
    pp.line_break()
    pp + f"Hypotheses per run: {len(descriptors)}" 
    pp + f"Max steps per run: {max_steps}" 
    pp + f"iterations: {iterations}" 
    pp.print()

    for run_index in range(iterations):
        print("Currently running iteration", run_index)
        descriptor_index = (run_index * (len(descriptors) - 1)) // max(iterations - 1, 1)
        simulated_room = next(room_supplier((descriptors[descriptor_index],)))
        simulated_robot = r3_model_robot(simulated_room)

        
        free_memory = getattr(gc, "mem_free", lambda: 0)
        allocated_memory = getattr(gc, "mem_alloc", lambda: 0)
        print("Before alg: Free RAM:", free_memory(), "bytes,", "Used RAM:", allocated_memory(), "bytes")
        alg = algorithm(simulated_robot, verbose=True, render=True)
        print("After alg: Free RAM:", free_memory(), "bytes,", "Used RAM:", allocated_memory(), "bytes")

        start_ms = timer.time()
        steps = 0
        finished = False
        while steps < max_steps and not finished:
            await alg.measure()
            finished = alg.step()
            steps += 1

        elapsed_ms = timer.time() - start_ms
        total_ms += elapsed_ms
        total_steps += steps
        if finished:
            finished_runs += 1
        print("Run %d: %d steps, %s, %.1f ms" % (
            run_index + 1,
            steps,
            "finished" if finished else "step limit",
            elapsed_ms,
        ))

        print("Before alg destruction: Free RAM:", free_memory(), "bytes,", "Used RAM:", allocated_memory(), "bytes")
        destroy_algorithm(alg)
        print("After alg destruction: Free RAM:", free_memory(), "bytes,", "Used RAM:", allocated_memory(), "bytes")


    pp = PrettyPrint()
    pp += f"Done running tests {iterations} tests!"
    
    pp.line_break()

    pp + f"Average run time: {(total_ms / iterations):.1f} ms"
    pp + f"Average time per step: {(total_ms / max(total_steps, 1)):.1f} ms"
    pp + f"Runs reaching an exit: {finished_runs} of {iterations}"
    pp.print()

class WrongPortError(Exception):
    __slots__ = ("port", "expected", "found", "_message")

    def __init__(self, port=None, expected=None, found=None, message=None):
        if message is None:
            if port is None and expected is None:
                message = "Wrong port"
            elif expected is None:
                message = "Wrong port: {}".format(port)
            elif port is None:
                message = "Wrong port, expected: {}".format(expected)
            elif found is None:
                message = "Wrong port: {}; expected {}".format(port, expected)
            else:
                message = "Wrong port: {}; expected {}, got {}".format(port, expected, found)

        self.port = port
        self.expected = expected
        self.found = found
        self._message = message

    def __str__(self):
        return self._message

    l = length(vec)
    if l == 0:
        return vec
    return tuple(v / l for v in vec)

def angle_between(a, b):
    la = length(a)
    lb = length(b)

    if la == 0 or lb == 0:
        return 0  # or handle differently

    cos_theta = dot(a, b) / (la * lb)

    # clamp for numerical safety
    cos_theta = max(-1, min(1, cos_theta))

    return acos(cos_theta)

def grid_neighbors(grid, coord, neighboorhood = STEP_TUPLE_FROM_DIR, default = 0):
    out = []
    for step in neighboorhood:
        ax = grid
        for dim, v in enumerate(step):
            d_pos = coord[dim] + v
            if d_pos < 0 or d_pos >= len(ax):
                ax = default
                break
            
            ax = ax[d_pos]

        out.append(ax)

    return out

def combinations(iterable, r):
    # combinations('ABCD', 2) --> AB AC AD BC BD CD
    # From itertools recipes
    pool = tuple(iterable)
    n = len(pool)
    if r > n:
        return
    indices = list(range(r))
    yield tuple(pool[i] for i in indices)
    while True:
        for i in reversed(range(r)):
            if indices[i] != i + n - r:
                break
        else:
            return
        indices[i] += 1
        for j in range(i + 1, r):
            indices[j] = indices[j - 1] + 1
        yield tuple(pool[i] for i in indices)

def possibilities(options, n):
    if n == 1:
        return [[v] for v in options]
    else:
        p = possibilities(options, n - 1)
        new = []
        for v in options:
            for pp in p:
                pp = pp.copy()
                pp.append(v)
                new.append(pp)
        return new

def entropy(counts):
    n = sum(counts)
    return -sum(c / n * log2(c / n) for c in counts if c)

S1 = expected score before room 3
S2 = expected score after room 3
E = probabilitie of exiting room 3
D = sum of probabilities of delivering first, second and third victim

A = follower points per second
L = room 3 start time
B = follower after room 3 expected time
T = time from start of room 3
60 = score for finishing follower

def update_r3_run_info(robot: r3_model_robot, info: r3_info, data: dict, action):
    data["visited_positions"] = list(data["visited_positions"])
    spread = info.spread_at(robot.x, robot.y)

    # Onboard and delivered victim info
    if not (robot.x, robot.y) in data["visited_positions"] and action == BACKWARD:
        # Only collect victims if not passed there before
        data["onboard_victims"] += spread["victim"]
    if action == FORWARD:
        data["delivered_victims"] += data["onboard_victims"] * spread["triangle"]
        data["onboard_victims"] *= 1 - spread["triangle"]

    data["visited_positions"].append((robot.x, robot.y))


def beam_state_key(candidate):
    robot = candidate[1]
    data = candidate[3]
    # Visit order and repeats do not matter: later logic only asks if a cell was visited.
    visited = tuple(sorted(set(data["visited_positions"])))
    # Exclude accumulated path score and first action; for an identical future state,
    # the beam should retain only the path with the higher score so far.
    return (
        robot.x,
        robot.y,
        robot.dir(),
        robot.time,
        robot.exited,
        data["onboard_victims"],
        data["delivered_victims"],
        visited,
        candidate[4],
        candidate[5],
    )


def insert_beam_candidate(beam, candidate, limit, seen_states):
    score = candidate[2]
    low = 0
    high = len(beam)
    while low < high:
        middle = (low + high) // 2
        if beam[middle][2] >= score:
            low = middle + 1
        else:
            high = middle

    if low >= limit:
        return
    key = beam_state_key(candidate)
    previous = seen_states.get(key)
    if previous is not None:
        # A lower-scoring path to the same future state cannot become the winner.
        if previous[2] >= score:
            return
        beam.remove(previous)
    beam.insert(low, candidate)
    seen_states[key] = candidate
    if len(beam) > limit:
        removed = beam.pop()
        removed_key = beam_state_key(removed)
        if seen_states.get(removed_key) is removed:
            del seen_states[removed_key]


# Decides to maximize score; what score to use will be hardcoded in a checklist.
# Will beam search
def decision_policy_2(
    real_robot: r3_model_robot,
    info: r3_info,
    persistent_data: dict,
    verbose=True,
    first_action_bonuses=None,
):
    if not info.room_descriptors:
        if verbose:
            print("[r3_algorithm] WARNING: no room hypotheses; turning left")
        return LEFT

    # ===================== [Beam search] =====================

    pa = [FORWARD, BACKWARD, LEFT, RIGHT]
    total_room_probability = sum(
        info.room_probability(descriptor)
        for descriptor in info.room_descriptors
    )
    BEAM = [(None, real_robot.copy(), 0, persistent_data.copy(), 0, total_room_probability)]
    beam_size = 165
    if verbose:
        print("[r3_algorithm] Running beam search...")
    for i in range(7):
        if verbose:
            print(" - [Beam search] at depth", i, "with", len(BEAM), "candidates")
        beam_size -= 20
        # Run beam search
        NEW_BEAM = []
        seen_states = {}
        for candidate in BEAM:
            for action in pa:
                # Unpack candidate data
                (candidate_first_movement, candidate_robot, candidate_score,
                 candidate_data, frozen_exit_score, active_probability) = candidate
                if candidate_first_movement is None: candidate_first_movement = action

                # Create new moved robot
                new_robot = candidate_robot.copy(); new_robot.noclip = True
                action_from_key(action, new_robot)

                # Disallow movements into a known collider
                if info.is_guaranteed_blocked(new_robot): continue

                candidate_data = candidate_data.copy()
                first_visit = (new_robot.x, new_robot.y) not in candidate_data["visited_positions"]
                update_r3_run_info(new_robot, info, candidate_data, action)

                # Compute weighted sum of score function for all rooms
                move_score = 0
                move_score += info_robot_score_function(candidate_data, info, new_robot)
                non_exit_score, exit_score = room_robot_score_functions(
                    new_robot,
                    candidate_data,
                )
                # Rooms sharing an exit cell all freeze their score on its first
                # visit. Track their combined probability instead of one bit per room.
                if first_visit:
                    probability = info.exit_probability_at(new_robot.x, new_robot.y)
                    frozen_exit_score += exit_score * probability
                    active_probability -= probability

                move_score += frozen_exit_score + non_exit_score * active_probability
                if i == 0 and first_action_bonuses is not None:
                    move_score += first_action_bonuses.get(action, 0)

                
                insert_beam_candidate(
                    NEW_BEAM,
                    (
                        candidate_first_movement,
                        new_robot,
                        candidate_score + move_score,
                        candidate_data,
                        frozen_exit_score,
                        active_probability,
                    ),
                    beam_size,
                    seen_states,
                )


        BEAM = NEW_BEAM
        if not BEAM:
            if verbose:
                print("[r3_algorithm] WARNING: no viable beam candidates; turning left")
            return LEFT
    
    winner = BEAM[0]
    return winner[0] # First movement of winner beam

def _measurement_for_room(robot, room, action):
    exited = robot.exited
    if action in MOVE_ACTIONS:
        kind = room.kind(robot.x + room.offset, robot.y)
        exited = kind if kind in ("exit", "entrance") else None

    distance = room.reading(

    touched_triangle = False

def _sensor_key(measurement):
    return tuple(measurement[index] for index in SENSOR_KEY_INDEXES)


class OpeningBookBuilder:
    def __init__(self, max_depth, max_nodes, information_weight):
        self.max_depth = max_depth
        self.max_nodes = max_nodes
        self.information_weight = information_weight
        self.nodes = [
            (ROOT_ACTION, []),
            (FALLBACK_ACTION, []),
            (TERMINAL_ACTION, []),
        ]
        self._canonical_nodes = {}
        self.decision_count = 0

    def _add_node(self, action, edges):
        edges = tuple(sorted(edges))
        signature = action, edges
        existing = self._canonical_nodes.get(signature)
        if existing is not None:
            return existing

        if len(self.nodes) >= self.max_nodes:
            raise RuntimeError(
                "opening book exceeded --max-nodes; lower --depth or raise the limit"
            )

        node_id = len(self.nodes)
        self.nodes.append((action, edges))
        self._canonical_nodes[signature] = node_id
        return node_id

    def _observation_groups(self, robot, descriptors, action=None):
        outcomes = {}
        sensor_groups = {}
        room = r3_model("3x3", 0, 1)

        for descriptor in descriptors:
            room_from_descriptor(descriptor, room)
            measurement = _measurement_for_room(robot, room, action)
            outcomes.setdefault(measurement, []).append(descriptor)
            sensor_groups.setdefault(_sensor_key(measurement), []).append(descriptor)

        return outcomes, sensor_groups

    def _information_bonuses(self, robot, info):
        bonuses = {}
        for action in ACTIONS:
            moved = robot.copy()
            moved.noclip = True
            action_from_key(action, moved)
            outcomes, _ = self._observation_groups(
                moved,
                info.room_descriptors,
                action,
            )

            # A room deterministically produces one measurement. Its outcome
            # entropy is therefore the expected information gained about the
            # hidden room by taking this action and measuring afterward.
            outcome_probabilities = []
            for descriptors in outcomes.values():
                probability = sum(
                    info.room_probability(descriptor)
                    for descriptor in descriptors
                )
                if probability > 0:
                    outcome_probabilities.append(probability)
            information_gain = -sum(
                probability * umath.log2(probability)
                for probability in outcome_probabilities
            )
            bonuses[action] = information_gain * self.information_weight
        return bonuses

    def _build_action_node(self, robot, info, persistent_data, action_depth):
        if action_depth >= self.max_depth or not info.room_descriptors:
            return 1

        information_bonuses = self._information_bonuses(robot, info)
        action = decision_policy_2(
            robot,
            info,
            persistent_data,
            verbose=False,
            first_action_bonuses=information_bonuses,
        )
        if action not in (FORWARD, BACKWARD, LEFT, RIGHT):
            return 1

        next_robot = robot.copy()
        next_robot.noclip = True
        action_from_key(action, next_robot)

        next_data = persistent_data.copy()
        update_r3_run_info(next_robot, info, next_data, action)

        outcomes, sensor_groups = self._observation_groups(
            next_robot,
            info.room_descriptors,
            action,
        )
        edges = []

        for measurement, _ in outcomes.items():
            if measurement[3] == "exit":
                child_id = 2
            else:
                descriptors = sensor_groups[_sensor_key(measurement)]
                child_info = r3_info(descriptors)
                child_robot = next_robot.copy()
                child_robot.exited = measurement[3]
                child_id = self._build_action_node(
                    child_robot,
                    child_info,
                    next_data,
                    action_depth + 1,
                )
            edges.append((encode_measurement(measurement), child_id))

        self.decision_count += 1
        return self._add_node(action, edges)

    def build(self):
        initial_info = r3_info()
        initial_robot = r3_model_robot(NULL_ROOM)
        initial_robot.noclip = True
        initial_robot.time = 0.0

        outcomes, sensor_groups = self._observation_groups(
            initial_robot,
            initial_info.room_descriptors,
        )
        root_edges = []

        for measurement in outcomes:
            descriptors = sensor_groups[_sensor_key(measurement)]
            child_info = r3_info(descriptors)
            child_robot = initial_robot.copy()
            child_robot.exited = measurement[3]
            child_id = self._build_action_node(
                child_robot,
                child_info,
                INITIAL_DATA.copy(),
                0,
            )
            root_edges.append((encode_measurement(measurement), child_id))

        self.nodes[0] = (ROOT_ACTION, tuple(sorted(root_edges)))
        return self.nodes

"""
COORDINATES (room frame)
    x goes right, y goes "up" into the room. The entrance wall is at y = -1.
    Room cells are 0 <= x < width, 0 <= y < height.
    The robot enters at cell (offset, 0).

DIRECTIONS    0 = up (+y)   1 = right (+x)   2 = down (-y)   3 = left (-x)
              turning right = +1, turning left = -1 (mod 4)

WALL NUMBERING: counterclockwise starting at the
    entrance wall. 0..W-1 bottom (left->right), then right wall going up,
    then top wall going right->left, then left wall going top->bottom.
    The entrance gap is wall number `offset`, the exit gap is wall `output`.

MISCELANEOUS
    * obstacle is 1 cell, or None
    * obstacle never sits right in front of the entrance or exit gap
    * a triangle sits in a corner cell, never on the obstacle, never next to a gap
    * sensor reading: 0..3 = number of free cells before something solid,
      4 = beam found nothing solid (it left through a gap)
"""

ROOM_SHAPE_IDS = {"3x3": 0, "3x4": 1, "4x3": 2}

# The room: pure data, never changes

# @dataclass(frozen=True)
class r3_model:
    shape: str                 # "3x3", "3x4" or "4x3"
    offset: int                # entrance gap = wall number `offset`
    output: int                # exit gap = wall number `output`
    obstacle: tuple = None     # (x, y) or None

    def __init__(self, shape, offset, output, obstacle=None):
        self.shape = shape
        self.offset = offset
        self.output = output
        self.obstacle = obstacle

    def width(self):  return SHAPES[self.shape][0]
    def height(self): return SHAPES[self.shape][1]

    def wall_cell(self, n):
        """Position (outside the room) of wall segment number n."""
        W, H = self.width(), self.height()
        if n < W: return (n, -1)
        n -= W
        if n < H: return (W, n)
        n -= H
        if n < W: return (W - 1 - n, H)
        n -= W
        return (-1, H - 1 - n)

    def entrance_cell(self): return self.wall_cell(self.offset)
    def exit_cell(self):     return self.wall_cell(self.output)
    def free_area(self):
        a = self.height() * self.width()
        a -= 2
        if not self.obstacle is None: a -= 1
        return a

    def corner_cell(self, c):
        W, H = self.width(), self.height()
        return [(0, 0), (W - 1, 0), (W - 1, H - 1), (0, H - 1)][c]

    def inside(self, x, y):
        return 0 <= x < self.width() and 0 <= y < self.height()

    def kind(self, x, y):
        """What is at (x, y)? 'free' | 'obstacle' | 'entrance' | 'exit' | 'wall'"""
        if self.inside(x, y):
            return "obstacle" if (x, y) == self.obstacle else "free"
        if (x, y) == self.entrance_cell(): return "entrance"
        if (x, y) == self.exit_cell():     return "exit"
        return "wall"

    def distance(self, x, y, d):
        """Walk from (x,y) in direction d. Returns (free_cells, what_stopped_us).
        Gaps are NOT solid: a beam through a gap just keeps going."""
        dx, dy = STEP_TUPLE_FROM_DIR[d]
        n = 0
        while True:
            x += dx; y += dy
            k = self.kind(x, y)
            if k == "free":
                n += 1
            else:
                return n, k

    def reading(self, x, y, d):
        """What the distance sensor reports."""
        n, k = self.distance(x, y, d)
        if k in ("entrance", "exit") or n >= SENSOR_FAR:
            return SENSOR_FAR
        return n

    def render(self, robot=None):
        W, H = self.width(), self.height()
        g = {}
        for y in range(-1, H + 1):
            for x in range(-1, W + 1):
                k = self.kind(x, y)
                g[(x, y)] = {"free": "  ", "obstacle": "[]", "wall": "â–ˆâ–ˆ",
                             "entrance": "^^", "exit": "--"}[k]
        from V3.r3_triangle_data import is_triangle
        for y in range(H):
            for x in range(W):
                if is_triangle(self, x, y):
                    g[(x, y)] = "TT"
        if robot is not None:
            g[(robot.x + self.offset, robot.y)] = "/>\\<"[robot.dir()] + "\\>/<"[robot.dir()]
        return "\n".join("".join(g[(x, y)] for x in range(-1, W + 1))
                         for y in range(H, -2, -1))

def room_descriptor(shape, offset, output, obstacle):
    if obstacle is None:
        obstacle_x, obstacle_y = NO_OBSTACLE
    else:
        obstacle_x, obstacle_y = obstacle

    return (
        obstacle_x
        | (obstacle_y << 2)
        | (ROOM_SHAPE_IDS[shape] << 4)
        | (offset << 6)
        | (output << 8)
    )


def room_from_descriptor(descriptor, reuse=None):
    obstacle_code = descriptor & OBSTACLE_MASK
    shape_id = (descriptor & SHAPE_MASK) >> 4
    offset = (descriptor & OFFSET_MASK) >> 6
    output = (descriptor & OUTPUT_MASK) >> 8
    obstacle = None
    if obstacle_code != 0b1111:
        obstacle = (obstacle_code & 0b11, obstacle_code >> 2)

    if reuse is None:
        return r3_model(ROOM_SHAPES[shape_id], offset, output, obstacle)

    reuse.shape = ROOM_SHAPES[shape_id]
    reuse.offset = offset
    reuse.output = output
    reuse.obstacle = obstacle
    return reuse


def room_kind_from_descriptor(x, y, descriptor):
    obstacle_code = descriptor & OBSTACLE_MASK
    shape_id = (descriptor & SHAPE_MASK) >> 4
    offset = (descriptor & OFFSET_MASK) >> 6
    width, height = SHAPES[ROOM_SHAPES[shape_id]]

    if 0 <= x < width and 0 <= y < height:
        if obstacle_code != 0b1111 and obstacle_code == (x | (y << 2)):
            return "obstacle"
        return "free"

    if x == offset and y == -1:
        return "entrance"

    if room_is_exit_from_descriptor(x, y, descriptor):
        return "exit"
    return "wall"


def room_is_exit_from_descriptor(x, y, descriptor):
    shape_id = (descriptor & SHAPE_MASK) >> 4
    width, height = SHAPES[ROOM_SHAPES[shape_id]]
    number = (descriptor & OUTPUT_MASK) >> 8
    if number < width:
        return x == number and y == -1
    number -= width
    if number < height:
        return x == width and y == number
    number -= height
    if number < width:
        return x == width - 1 - number and y == height
    number -= width
    return x == -1 and y == height - 1 - number


def room_frequency_score(descriptor):
    obstacle_code = descriptor & OBSTACLE_MASK
    shape_id = (descriptor & SHAPE_MASK) >> 4
    if obstacle_code == 0b1111:
        obstacle_probability = 1 - probabilitie_of_room3_having_a_obstacle
    else:
        obstacle_probability = probabilitie_of_room3_having_a_obstacle
    return obstacle_probability * probabilitie_of_room3_per_shape[ROOM_SHAPES[shape_id]]

# Every legal room

def legal_triangle_pairs(shape, offset, output, obstacle):
    base = r3_model(shape, offset, output, obstacle)
    gaps = [base.entrance_cell(), base.exit_cell()]
    free_corners = []
    for corner in range(4):
        cx, cy = base.corner_cell(corner)
        if (cx, cy) == obstacle:
            continue
        if any(abs(gx - cx) + abs(gy - cy) == 1 for gx, gy in gaps):
            continue
        free_corners.append(corner)
    return [(a, b) for a, b in combinations(free_corners, 2)]


def descriptor_triangle_pairs(descriptor):
    obstacle_code = descriptor & OBSTACLE_MASK
    shape_id = (descriptor & SHAPE_MASK) >> 4
    offset = (descriptor & OFFSET_MASK) >> 6
    output = (descriptor & OUTPUT_MASK) >> 8
    obstacle = None if obstacle_code == 0b1111 else (obstacle_code & 0b11, obstacle_code >> 2)
    return legal_triangle_pairs(ROOM_SHAPES[shape_id], offset, output, obstacle)


def descriptor_triangle_probability_at(descriptor, x, y):
    shape_id = (descriptor & SHAPE_MASK) >> 4
    offset = (descriptor & OFFSET_MASK) >> 6
    output = (descriptor & OUTPUT_MASK) >> 8
    obstacle_code = descriptor & OBSTACLE_MASK
    obstacle = None if obstacle_code == 0b1111 else (obstacle_code & 0b11, obstacle_code >> 2)
    room = r3_model(ROOM_SHAPES[shape_id], offset, output, obstacle)
    pairs = legal_triangle_pairs(room.shape, offset, output, obstacle)
    if not pairs:
        return 0
    matches = sum(1 for pair in pairs if any(room.corner_cell(c) == (x, y) for c in pair))
    return matches / len(pairs)


def descriptor_can_have_triangle_at(descriptor, x, y):
    shape_id = (descriptor & SHAPE_MASK) >> 4
    offset = (descriptor & OFFSET_MASK) >> 6
    room = r3_model(ROOM_SHAPES[shape_id], offset, (descriptor & OUTPUT_MASK) >> 8,
                    None if (descriptor & OBSTACLE_MASK) == 0b1111 else
                    (descriptor & 0b11, (descriptor & OBSTACLE_MASK) >> 2))
    return any(room.corner_cell(c) == (x, y) for pair in descriptor_triangle_pairs(descriptor) for c in pair)


def _iter_room_descriptors():
    for shape, (W, H) in SHAPES.items():
        for off in range(W):                       # entrance is on the bottom wall
            for out in range(2 * (W + H)):
                if out == off:
                    continue
                base = r3_model(shape, off, out)
                gaps = [base.entrance_cell(), base.exit_cell()]

                def in_front(g):                   # room cell right inside a gap
                    return (min(max(g[0], 0), W - 1), min(max(g[1], 0), H - 1))
                blocked = {in_front(g) for g in gaps}

                if ONLY_OBSTACLE_LESS_ROOMS:
                    obstacles = [None]
                else:
                    obstacles = [None] + [(x, y) for x in range(W) for y in range(H)
                                          if (x, y) not in blocked]
                for obs in obstacles:
                    yield room_descriptor(shape, off, out, obs)


def all_room_descriptors():
    return list(_iter_room_descriptors())

def room_supplier(descriptors):
    room = r3_model("3x3", 0, 1)
    from V3.r3_triangle_data import set_triangles
    for descriptor in descriptors:
        room_from_descriptor(descriptor, room)
        pairs = descriptor_triangle_pairs(descriptor)
        set_triangles(room, pairs[0] if pairs else ())
        yield room

        n.log             =    self.log
        return n


    def _turn(self, s):
        self._dir = (self._dir + s) % 4
        self.garra_touched_wall = False
        self.touched_triangle = False
        self._pay("turn")

    def forward(self):
        """Move one cell. Returns False if blocked (wall/obstacle)."""
        self._pay("move")
        self.garra_touched_wall = False
        self.touched_triangle = False

        self.last_dir = (dx, dy)

        self.garra_touched_wall = k in ("wall", "obstacle")
        if self.garra_touched_wall:
            return False

        self.x, self.y = nx, ny
        
    
        if k in ("exit", "entrance"):
            self.exited = k
        return True

    def backyard(self):
        """Move one cell back; triangle contact restores the original tile."""
        self._pay("move")
        self.garra_touched_wall = False
        self.touched_triangle = False
        dx, dy = STEP_TUPLE_FROM_DIR[self._dir]
        self.last_dir = (-dx, -dy)

        k = "free" if self.noclip else self._room.kind(nx + self._room.offset, ny)
        if k in ("wall", "obstacle"):

        if self.on_triangle(room=self._room):

def encode_measurement(measurement):
    distance, touched_triangle, touched_wall, exited = measurement
    if not 0 <= distance <= 4:
        raise ValueError("distance must be from 0 to 4")

    exit_code = {None: 0, "exit": 1, "entrance": 2}.get(exited)
    if exit_code is None:
        raise ValueError("invalid exit measurement")
    return (
        distance
        | (int(bool(touched_triangle)) << 3)
        | (int(bool(touched_wall)) << 4)
        | (exit_code << 5)
    )


def decode_measurement(code):
    distance = code & 0x07
    touched_triangle = bool((code >> 3) & 1)
    touched_wall = bool((code >> 4) & 1)
    exit_code = (code >> 5) & 0x03

    if distance > 4 or exit_code > 2:
        raise ValueError("invalid encoded measurement")
    return distance, touched_triangle, touched_wall, (None, "exit", "entrance")[exit_code]


def encode_nodes(nodes):
    count = len(nodes)
    if count > 0xFFFF:
        raise ValueError("too many opening-book nodes")

    records = []
    offsets = []
    cursor = 7 + count * 2
    for action, edges in nodes:
        if not 0 <= action <= 0xFF or len(edges) > 0xFF:
            raise ValueError("node action or edge count is out of range")
        record = bytearray((action, len(edges)))
        for measurement_code, child in edges:
            if not 0 <= measurement_code <= 0xFF or not 0 <= child < count:
                raise ValueError("invalid opening-book edge")
            record.append(measurement_code)
            record.append(child & 0xFF)
            record.append((child >> 8) & 0xFF)
        offsets.append(cursor)
        records.append(record)
        cursor += len(record)

    if cursor > 0xFFFF:
        raise ValueError("opening book exceeds the 16-bit format limit")

    data = bytearray(b"R3OB")
    data.append(BOOK_FORMAT_VERSION)
    data.append(count & 0xFF)
    data.append((count >> 8) & 0xFF)
    for offset in offsets:
        data.append(offset & 0xFF)
        data.append((offset >> 8) & 0xFF)
    for record in records:
        data.extend(record)
    return bytes(data)


class OpeningBook:
    def __init__(self, data):
        if len(data) < 7 or data[0] != 82 or data[1] != 51 or data[2] != 79 or data[3] != 66:
            raise ValueError("invalid opening-book header")
        if data[4] != BOOK_FORMAT_VERSION:
            raise ValueError("unsupported opening-book version")

        self.data = data
        self.node_count = data[5] | (data[6] << 8)
        self.offset_table_end = 7 + self.node_count * 2
        if self.offset_table_end > len(data):
            raise ValueError("truncated opening-book offsets")

    def _node_offset(self, node_id):
        if node_id < 0 or node_id >= self.node_count:
            return None
        index = 7 + node_id * 2
        offset = self.data[index] | (self.data[index + 1] << 8)
        if offset < self.offset_table_end or offset + 2 > len(self.data):
            return None
        return offset

    def action_for(self, node_id):
        offset = self._node_offset(node_id)
        if offset is None:
            return None
        action = self.data[offset]
        return action if action < 4 else None

    def next_node(self, node_id, measurement):
        offset = self._node_offset(node_id)
        if offset is None:
            return None

        code = encode_measurement(measurement)
        edge_count = self.data[offset + 1]
        edge = offset + 2
        for _ in range(edge_count):
            if edge + 2 >= len(self.data):
                return None
            child = self.data[edge + 1] | (self.data[edge + 2] << 8)
            if self.data[edge] == code:
                return child if child < self.node_count else None
            edge += 3
        return None

def room_robot_score_functions(robot: r3_model_robot, candidate_data: dict):
    delivered_victims = candidate_data["delivered_victims"]
    onboard_victims = candidate_data["onboard_victims"]

    time_from_round_beggining = avg_time_before_r3 + robot.time
    time_to_round_end = round_duration - time_from_round_beggining

    """
    In order to know what score we'll be gaining by delivering the victims, we 
    need to know the total score to multiply. Thats why we calculate the expected score
    after r3! Furthermore, if by delivering the victims the robot exits too late, so
    that we get few points after r3, them it gets less rewarded!
    """

    potencial_score_after_r3 = avg_score_after_r3 * saturate(time_to_round_end, avg_time_after_r3 * 0.2)
    # potencial_score_after_r3 += 60 * nk.sigmoid(time_to_round_end - avg_time_after_r3, 3)
    expected_score = avg_score_before_r3 + potencial_score_after_r3
    expected_score_multiplier = pow(score_multiplier_per_victim, delivered_victims)

    score_before_onboard = expected_score * expected_score_multiplier
    non_exit_score = score_before_onboard * pow(
        score_multiplier_per_victim - 0.05,
        onboard_victims,
    )
    exit_score = score_before_onboard + 10 * sigmoid(delivered_victims - 3)
    return non_exit_score, exit_score

def set_triangles(room, triangles):
    """Associate triangles with a room without external runtime dependencies."""
    for index, entry in enumerate(_triangle_entries):
        if entry[0] is room:
            _triangle_entries[index] = (room, tuple(triangles))
            return
    _triangle_entries.append((room, tuple(triangles)))


def triangles_for(room):
    for candidate, triangles in _triangle_entries:
        if candidate is room:
            return triangles
    return ()


def is_triangle(room, x, y):
    for corner in triangles_for(room):
        if room.corner_cell(corner) == (x, y):
            return True
    return False

        byte b = packet[i];

            UART1.write(ALT_FLAG);
            UART1.write(ALT_ALT_FLAG);

void _add_byte_to_buffer(byte b) {
    if (pointer >= 256)
        return;

    packetBuffer[packetBufferIndex][pointer] = b;
    pointer++;
}

byte unalt(byte b) {
    if (b == ALT_ALT_FLAG)
        return ALT_FLAG;

    if (b == ALT_STOP_FLAG)
        return STOP_FLAG;

    return b;
}


void tick_listener() {
    while (UART1.available()) {
        byte b = UART1.read();

        if (b == STOP_FLAG) {
            pointer = 0;
            packetBufferIndex = (packetBufferIndex + 1) % 8;
            is_packet_new = true;
        }
        else if (is_alt) {
            _add_byte_to_buffer(unalt(b));
            is_alt = false;
        }
        else if (b == ALT_FLAG) {
            is_alt = true;
        }
        else {
            _add_byte_to_buffer(b);
        }
    }
}

void setup() {


  // Startup logging
  Serial.begin(115200);
  pinMode(ledR, OUTPUT);
  pinMode(ledG, OUTPUT);
  pinMode(ledB, OUTPUT);
  pinMode(rear_button_pin_L, INPUT);
  pinMode(rear_button_pin_R, INPUT);
  digitalWrite(ledR, LOW);
  digitalWrite(ledG, LOW);
  digitalWrite(ledB, LOW);
  Serial.println("ESP32 Hub alive!");
  Serial.println("===============[ Info ]===============");
  Serial.println(ESP.getChipModel());
  Serial.println(ESP.getChipRevision());
  Serial.println(ESP.getChipCores());
  Serial.println("======================================");

  // Hub bridge startup
  UART1.begin(115200, SERIAL_8N1, 16, 17);
  startup_distance_sensor();

}

void rgb_log(float r, float g, float b){
  analogWrite(ledR, 254 * r);
  analogWrite(ledG, 254 * g);
  analogWrite(ledB, 254 * b);
}


void rgb_sine_log(float r, float g, float b, float f){
  float s = (sin(f * millis() * 0.001 * 3.1415) + 1) * 0.5;
  rgb_log(r * s, g * s, b * s);
}

void rgb_sine_log_delay(float r, float g, float b, float f, int dd){
  for (int i = 0; i < dd; i+= 2){
    rgb_sine_log(r, g, b, f);
    delay(2);
  }
}




void loop() {
    static uint16_t last_valid_distance = 0;

    uint16_t distance = sensor.read();

    byte packet[4];
    float r = 0;
    float b = 0;
    byte rear_button_L = 0x00;
    byte rear_button_R = 0x00;
\
    if (digitalRead(rear_button_pin_L) == LOW){
      Serial.print("L-on  "); 
      rear_button_L = 0x01; 
      r = 0.3;
    } else {
      Serial.print("L-off ");
    }
    if (digitalRead(rear_button_pin_R) == LOW){
      Serial.println("R-on"); 
      rear_button_R = 0x01; 
      b = 0.3;
    } else {
      Serial.println("R-off");
    }

    if (sensor.timeoutOccurred()) {
        // Keep the previous valid distance
        packet[0] = lowByte(last_valid_distance);
        packet[1] = highByte(last_valid_distance);
        packet[2] = 0x01; // TIMEOUT
        packet[3] = rear_button_L;
        packet[4] = rear_button_R;

        Serial.println("TIMEOUT");

        send_packet(packet, 5);
        restart_distance_sensor();
    }
    else {
        last_valid_distance = distance;

        packet[0] = lowByte(distance);
        packet[1] = highByte(distance);
        packet[2] = 0x00; // VALID
        packet[3] = rear_button_L;
        packet[4] = rear_button_R;

        //Serial.println(distance);

        send_packet(packet, 5);
    }
    //rgb.log(float(int(packetBuffer[(packetBufferIndex + 7) % 8][0])) / 254, float(int(packetBuffer[(packetBufferIndex + 7) % 8][1])) / 254, float(int(packetBuffer[(packetBufferIndex + 7) % 8][2])) / 254)
    
    rgb_sine_log_delay(r, 0.5, b, 1, 100);
    tick_listener();
    if (is_packet_new){
      is_packet_new = false;
      send_packet(packetBuffer[(packetBufferIndex + 7) % 8], 10);
      rgb_log(0, 0, 1);
      delay(1000);
    }
}
